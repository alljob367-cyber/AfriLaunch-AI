   521→   521→// Deep merge for nested config objects
   522→   522→function deepMerge<T>(target: T, source: Partial<T>): T {
   523→   523→  if (typeof target !== 'object' || target === null) return (source ?? target) as T;
   524→   524→  if (typeof source !== 'object' || source === null) return target;
   525→   525→  const result: any = Array.isArray(target) ? [...(target as any)] : { ...target };
   526→   526→  for (const key of Object.keys(source as any)) {
   527→   527→    const srcVal = (source as any)[key];
   528→   528→    const tgtVal = (target as any)[key];
   529→   529→    if (srcVal && typeof srcVal === 'object' && !Array.isArray(srcVal) && tgtVal && typeof tgtVal === 'object') {
   530→   530→      result[key] = deepMerge(tgtVal, srcVal);
   531→   531→    } else if (srcVal !== undefined) {
   532→   532→      result[key] = srcVal;
   533→   533→    }
   534→   534→  }
   535→   535→  return result as T;
   536→   536→}

   572→   572→export async function createSession(expiryHours: number): Promise<string> {
   573→   573→  const token = crypto.randomBytes(32).toString('hex');
   574→   574→  const now = Date.now();
   575→   575→  const session: AdminSession = {
   576→   576→    token,
   577→   577→    createdAt: now,
   578→   578→    expiresAt: now + expiryHours * 60 * 60 * 1000,
   579→   579→  };
   580→   580→  const sessions = await readSessions();
   581→   581→  sessions.push(session);
   582→   582→  await writeSessions(sessions);
   583→   583→  return token;
   584→   584→}

   586→   586→export async function validateSession(token: string): Promise<boolean> {
   587→   587→  const sessions = await readSessions();
   588→   588→  const now = Date.now();
   589→   589→  const valid = sessions.find((s) => s.token === token && s.expiresAt > now);
   590→   590→  if (!valid) return false;
   591→   591→  // Cleanup expired sessions
   592→   592→  const active = sessions.filter((s) => s.expiresAt > now);
   593→   593→  if (active.length !== sessions.length) await writeSessions(active);
   594→   594→  return true;
   595→   595→}
