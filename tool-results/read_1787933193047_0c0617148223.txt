    57→async function pollJob(jobId: string, onStatus?: (status: string, elapsed: number) => void): Promise<{ ok: boolean; content?: string; error?: string; provider?: string; model?: string }> {
    58→  const maxAttempts = 100; // 100 × 3s = 5 min max
    59→  for (let i = 0; i < maxAttempts; i++) {
    60→    await new Promise(r => setTimeout(r, 3000));
    61→    try {
    62→      const res = await fetch(`/api/ai/generate-async?jobId=${jobId}`, { credentials: 'include' });
    63→      const data = await res.json();
    64→      if (onStatus) onStatus(data.status, data.elapsed || 0);
    65→      if (data.status === 'done' && data.result) {
    66→        return { ok: true, content: data.result.content, provider: data.result.provider, model: data.result.model };
    67→      }
    68→      if (data.status === 'failed') {
    69→        return { ok: false, error: data.error || 'Génération échouée' };
    70→      }
    71→      // Still pending or running → continue polling
    72→    } catch { /* retry */ }
    73→  }
    74→  return { ok: false, error: 'Timeout: la génération prend trop de temps' };
    75→}

   128→  async function handleGenerate() {

   141→        credentials: 'include',
   142→        body: JSON.stringify({ type: 'content', format, topic, businessName, industry, audience, tone, batch }),
   143→      });
   144→      const data = await res.json();
   145→      if (!data.ok) {
   146→        if (data.insufficientCredits) {
   147→          toast({ title: 'Crédits insuffisants', description: `La génération coûte ${creditCost} crédits.`, variant: 'error' });
   148→        } else if (data.dailyLimitReached) {
   149→          toast({ title: 'Limite quotidienne atteinte', description: data.error, variant: 'error' });
   150→        } else {
   151→          toast({ title: 'Échec', description: data.error, variant: 'error' });
   152→        }
   153→        setGenerating(false);
   154→        setStatusMsg('');
   155→        return;
   156→      }
   157→      const creditsRemaining = data.creditsRemaining;
   158→      // Poll the async job until done or failed
   159→      const startTime = Date.now();
   160→      const result = await pollJob(data.jobId, (_status, _elapsed) => {
   161→        const seconds = Math.max(1, Math.round((Date.now() - startTime) / 1000));
   162→        setStatusMsg(`Génération en cours... (${seconds}s)`);
   163→      });
   164→      if (!result.ok) {
   165→        toast({ title: 'Échec', description: result.error, variant: 'error' });
   166→        setGenerating(false);
   167→        setStatusMsg('');
   168→        return;
   169→      }
   170→      const content = result.content || '';
   171→      const pieces = batch
   172→        ? content.split('---VARIANTE---').map((s: string) => s.trim()).filter(Boolean)
   173→        : [content.trim()];
   174→      setResults(pieces);
   175→      if (typeof creditsRemaining === 'number') setCreditsRemaining(creditsRemaining);
   176→      toast({ title: `${pieces.length} contenu(s) généré(s) ! ✍️`, description: `${creditCost} crédits débités`, variant: 'success' });
   177→
   178→      // Generate image if applicable
   179→      if (generateImage && showImageOption && !batch) {
   180→        generateImageForContent(pieces[0]);
   181→      }
   182→    } catch (err) {
   183→      toast({ title: 'Erreur réseau', description: (err as Error).message, variant: 'error' });
   184→    } finally {
   185→      setGenerating(false);
   186→      setStatusMsg('');
   187→    }
   188→  }

   190→  async function generateImageForContent(contentText: string) {
   191→    setGeneratingImage(true);
   192→    try {
   193→      // Build a visual prompt from the content + business context
   194→      const promptParts = [
   195→        businessName || 'business',
   196→        industry || '',
   197→        topic || '',
   198→      ].filter(Boolean).join(', ');
   199→      // Use the AI to generate a short visual prompt
   200→      const visualPromptRes = await fetch('/api/ai/generate', {
   201→        method: 'POST',
   202→        headers: { 'Content-Type': 'application/json' },
   203→        credentials: 'include',
   204→        body: JSON.stringify({
   205→          type: 'content',
   206→          format: 'ad-copy',
   207→          topic: `Génère une description visuelle courte (max 50 mots) pour une image de post réseaux sociaux: ${promptParts}. Décris la scène, les couleurs, le style.`,
   208→          businessName,
   209→          industry,
   210→          tone: 'Commercial',
   211→        }),
   212→      });
   213→      const visualData = await visualPromptRes.json();
   214→      let visualPrompt = '';
   215→      if (visualData.ok) {
   216→        visualPrompt = visualData.content.trim().slice(0, 200);
   217→      } else {
   218→        visualPrompt = `${businessName} ${industry} ${topic} professional social media post`.slice(0, 200);
   219→      }
   220→
   221→      // Generate image via Pollinations.ai (free, no API key)
   222→      const encodedPrompt = encodeURIComponent(visualPrompt);
   223→      const seed = Math.floor(Math.random() * 1000000);
   224→      const imageUrl = `https://image.pollinations.ai/prompt/${encodedPrompt}?width=1080&height=1080&seed=${seed}&nologo=true&model=flux`;
   225→      setGeneratedImageUrl(imageUrl);
   226→      toast({ title: 'Image générée ! 🖼️', description: 'Image visuelle créée pour votre post', variant: 'success' });
   227→    } catch {
   228→      toast({ title: 'Image non générée', description: 'Le texte a été généré avec succès.', variant: 'warning' });
   229→    } finally {
   230→      setGeneratingImage(false);
   231→    }
   232→  }
