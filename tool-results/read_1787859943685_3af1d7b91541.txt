   481→// ─── File I/O ─────────────────────────────────────────────────────────
   482→async function readConfig(): Promise<AppConfig> {
   483→  try {
   484→    const raw = await fs.readFile(CONFIG_PATH, 'utf-8');
   485→    const parsed = JSON.parse(raw) as Partial<AppConfig>;
   486→    // Merge with defaults to handle schema evolution
   487→    const defaults = getDefaultConfig();
   488→    return { ...defaults, ...parsed } as AppConfig;
   489→  } catch (err: unknown) {
   490→    // File doesn't exist yet — create with defaults
   491→    const defaults = getDefaultConfig();
   492→    await writeConfig(defaults);
   493→    return defaults;
   494→  }
   495→}
   496→
   497→async function writeConfig(config: AppConfig): Promise<void> {
   498→  config.updatedAt = new Date().toISOString();
   499→  config.version += 1;
   500→  await fs.mkdir(path.dirname(CONFIG_PATH), { recursive: true });
   501→  await fs.writeFile(CONFIG_PATH, JSON.stringify(config, null, 2), 'utf-8');
   502→}
   503→
   504→// ─── Public API ───────────────────────────────────────────────────────
   505→let configCache: AppConfig | null = null;
   506→
   507→export async function getConfig(): Promise<AppConfig> {
   508→  if (configCache) return configCache;
   509→  configCache = await readConfig();
   510→  return configCache;
   511→}
   512→
   513→export async function updateConfig(updates: Partial<AppConfig>): Promise<AppConfig> {
   514→  const current = await getConfig();
   515→  const next = deepMerge(current, updates);
   516→  await writeConfig(next);
   517→  configCache = next;
   518→  return next;
   519→}

   521→// Deep merge for nested config objects
   522→function deepMerge<T>(target: T, source: Partial<T>): T {
   523→  if (typeof target !== 'object' || target === null) return (source ?? target) as T;
   524→  if (typeof source !== 'object' || source === null) return target;
   525→  const result: any = Array.isArray(target) ? [...(target as any)] : { ...target };
   526→  for (const key of Object.keys(source as any)) {
   527→    const srcVal = (source as any)[key];
   528→    const tgtVal = (target as any)[key];
   529→    if (srcVal && typeof srcVal === 'object' && !Array.isArray(srcVal) && tgtVal && typeof tgtVal === 'object') {
   530→      result[key] = deepMerge(tgtVal, srcVal);
   531→    } else if (srcVal !== undefined) {
   532→      result[key] = srcVal;
   533→    }
   534→  }
   535→  return result as T;
   536→}

   558→async function readSessions(): Promise<AdminSession[]> {
   559→  try {
   560→    const raw = await fs.readFile(SESSIONS_PATH, 'utf-8');

   562→  } catch {
   563→    return [];

   567→async function writeSessions(sessions: AdminSession[]): Promise<void> {
   568→  await fs.mkdir(path.dirname(SESSIONS_PATH), { recursive: true });
   569→  await fs.writeFile(SESSIONS_PATH, JSON.stringify(sessions, null, 2), 'utf-8');
   570→}

   572→export async function createSession(expiryHours: number): Promise<string> {
   573→  const token = crypto.randomBytes(32).toString('hex');
   574→  const now = Date.now();
   575→  const session: AdminSession = {
   576→    token,
   577→    createdAt: now,
   578→    expiresAt: now + expiryHours * 60 * 60 * 1000,
   579→  };
   580→  const sessions = await readSessions();
   581→  sessions.push(session);
   582→  await writeSessions(sessions);
   583→  return token;
   584→}

   586→export async function validateSession(token: string): Promise<boolean> {
   587→  const sessions = await readSessions();
   588→  const now = Date.now();
   589→  const valid = sessions.find((s) => s.token === token && s.expiresAt > now);
   590→  if (!valid) return false;
   591→  // Cleanup expired sessions
   592→  const active = sessions.filter((s) => s.expiresAt > now);
   593→  if (active.length !== sessions.length) await writeSessions(active);
   594→  return true;
   595→}
   596→
   597→export async function destroySession(token: string): Promise<void> {
   598→  const sessions = await readSessions();
   599→  const remaining = sessions.filter((s) => s.token !== token);
   600→  await writeSessions(remaining);
   601→}
