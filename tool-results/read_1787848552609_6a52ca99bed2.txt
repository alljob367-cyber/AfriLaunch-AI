   462→// ─── File I/O ─────────────────────────────────────────────────────────
   463→async function readConfig(): Promise<AppConfig> {
   464→  try {
   465→    const raw = await fs.readFile(CONFIG_PATH, 'utf-8');
   466→    const parsed = JSON.parse(raw) as Partial<AppConfig>;
   467→    // Merge with defaults to handle schema evolution
   468→    const defaults = getDefaultConfig();
   469→    return { ...defaults, ...parsed } as AppConfig;
   470→  } catch (err: unknown) {
   471→    // File doesn't exist yet — create with defaults
   472→    const defaults = getDefaultConfig();
   473→    await writeConfig(defaults);
   474→    return defaults;
   475→  }
   476→}
   477→
   478→async function writeConfig(config: AppConfig): Promise<void> {
   479→  config.updatedAt = new Date().toISOString();
   480→  config.version += 1;
   481→  await fs.mkdir(path.dirname(CONFIG_PATH), { recursive: true });
   482→  await fs.writeFile(CONFIG_PATH, JSON.stringify(config, null, 2), 'utf-8');
   483→}

   486→let configCache: AppConfig | null = null;

   488→export async function getConfig(): Promise<AppConfig> {
   489→  if (configCache) return configCache;
   490→  configCache = await readConfig();
   491→  return configCache;
   492→}

   494→export async function updateConfig(updates: Partial<AppConfig>): Promise<AppConfig> {
   495→  const current = await getConfig();
   496→  const next = deepMerge(current, updates);
   497→  await writeConfig(next);
   498→  configCache = next;
   499→  return next;
   500→}

   502→// Deep merge for nested config objects
   503→function deepMerge<T>(target: T, source: Partial<T>): T {
   504→  if (typeof target !== 'object' || target === null) return (source ?? target) as T;
   505→  if (typeof source !== 'object' || source === null) return target;
   506→  const result: any = Array.isArray(target) ? [...(target as any)] : { ...target };
   507→  for (const key of Object.keys(source as any)) {
   508→    const srcVal = (source as any)[key];
   509→    const tgtVal = (target as any)[key];
   510→    if (srcVal && typeof srcVal === 'object' && !Array.isArray(srcVal) && tgtVal && typeof tgtVal === 'object') {
   511→      result[key] = deepMerge(tgtVal, srcVal);
   512→    } else if (srcVal !== undefined) {
   513→      result[key] = srcVal;
   514→    }
   515→  }
   516→  return result as T;
   517→}

   519→// ─── Password hashing ─────────────────────────────────────────────────
   520→export function hashPassword(password: string): string {
   521→  return crypto.createHash('sha256').update(password).digest('hex');
   522→}

   532→// ─── Session management ───────────────────────────────────────────────
   533→interface AdminSession {
   534→  token: string;
   535→  createdAt: number;
   536→  expiresAt: number;
   537→}

   539→async function readSessions(): Promise<AdminSession[]> {
   540→  try {
   541→    const raw = await fs.readFile(SESSIONS_PATH, 'utf-8');
   542→    return JSON.parse(raw) as AdminSession[];
   543→  } catch {
   544→    return [];
   545→  }
   546→}
   547→
   548→async function writeSessions(sessions: AdminSession[]): Promise<void> {
   549→  await fs.mkdir(path.dirname(SESSIONS_PATH), { recursive: true });
   550→  await fs.writeFile(SESSIONS_PATH, JSON.stringify(sessions, null, 2), 'utf-8');
   551→}

   553→export async function createSession(expiryHours: number): Promise<string> {
   554→  const token = crypto.randomBytes(32).toString('hex');
   555→  const now = Date.now();
   556→  const session: AdminSession = {
   557→    token,
   558→    createdAt: now,
   559→    expiresAt: now + expiryHours * 60 * 60 * 1000,
   560→  };
   561→  const sessions = await readSessions();
   562→  sessions.push(session);
   563→  await writeSessions(sessions);
   564→  return token;
   565→}
   566→
   567→export async function validateSession(token: string): Promise<boolean> {
   568→  const sessions = await readSessions();
   569→  const now = Date.now();
   570→  const valid = sessions.find((s) => s.token === token && s.expiresAt > now);
   571→  if (!valid) return false;
   572→  // Cleanup expired sessions
   573→  const active = sessions.filter((s) => s.expiresAt > now);
   574→  if (active.length !== sessions.length) await writeSessions(active);
   575→  return true;
   576→}
   577→
   578→export async function destroySession(token: string): Promise<void> {
   579→  const sessions = await readSessions();
   580→  const remaining = sessions.filter((s) => s.token !== token);
   581→  await writeSessions(remaining);
   582→}
