   434→// ─── File I/O ─────────────────────────────────────────────────────────
   435→async function readConfig(): Promise<AppConfig> {
   436→  try {
   437→    const raw = await fs.readFile(CONFIG_PATH, 'utf-8');
   438→    const parsed = JSON.parse(raw) as Partial<AppConfig>;
   439→    // Merge with defaults to handle schema evolution
   440→    const defaults = getDefaultConfig();
   441→    return { ...defaults, ...parsed } as AppConfig;
   442→  } catch (err: unknown) {
   443→    // File doesn't exist yet — create with defaults
   444→    const defaults = getDefaultConfig();
   445→    await writeConfig(defaults);
   446→    return defaults;
   447→  }
   448→}
   449→
   450→async function writeConfig(config: AppConfig): Promise<void> {
   451→  config.updatedAt = new Date().toISOString();
   452→  config.version += 1;
   453→  await fs.mkdir(path.dirname(CONFIG_PATH), { recursive: true });
   454→  await fs.writeFile(CONFIG_PATH, JSON.stringify(config, null, 2), 'utf-8');
   455→}

   460→export async function getConfig(): Promise<AppConfig> {
   461→  if (configCache) return configCache;
   462→  configCache = await readConfig();

   466→export async function updateConfig(updates: Partial<AppConfig>): Promise<AppConfig> {
   467→  const current = await getConfig();
   468→  const next = deepMerge(current, updates);
   469→  await writeConfig(next);
   470→  configCache = next;
   471→  return next;
   472→}

   474→// Deep merge for nested config objects
   475→function deepMerge<T>(target: T, source: Partial<T>): T {
   476→  if (typeof target !== 'object' || target === null) return (source ?? target) as T;
   477→  if (typeof source !== 'object' || source === null) return target;
   478→  const result: any = Array.isArray(target) ? [...(target as any)] : { ...target };
   479→  for (const key of Object.keys(source as any)) {
   480→    const srcVal = (source as any)[key];
   481→    const tgtVal = (target as any)[key];
   482→    if (srcVal && typeof srcVal === 'object' && !Array.isArray(srcVal) && tgtVal && typeof tgtVal === 'object') {
   483→      result[key] = deepMerge(tgtVal, srcVal);
   484→    } else if (srcVal !== undefined) {
   485→      result[key] = srcVal;
   486→    }
   487→  }
   488→  return result as T;
   489→}

   491→// ─── Password hashing ─────────────────────────────────────────────────
   492→export function hashPassword(password: string): string {
   493→  return crypto.createHash('sha256').update(password).digest('hex');
   494→}
   495→
   496→export function verifyPassword(password: string, hash: string | null): boolean {
   497→  if (!hash) {
   498→    // First-time setup: default password is "admin123"
   499→    return password === 'admin123';
   500→  }
   501→  return hashPassword(password) === hash;
   502→}

   525→export async function createSession(expiryHours: number): Promise<string> {
   526→  const token = crypto.randomBytes(32).toString('hex');
   527→  const now = Date.now();
   528→  const session: AdminSession = {
   529→    token,
   530→    createdAt: now,
   531→    expiresAt: now + expiryHours * 60 * 60 * 1000,
   532→  };
   533→  const sessions = await readSessions();
   534→  sessions.push(session);
   535→  await writeSessions(sessions);
   536→  return token;
   537→}

   539→export async function validateSession(token: string): Promise<boolean> {
   540→  const sessions = await readSessions();
   541→  const now = Date.now();
   542→  const valid = sessions.find((s) => s.token === token && s.expiresAt > now);
   543→  if (!valid) return false;
   544→  // Cleanup expired sessions
   545→  const active = sessions.filter((s) => s.expiresAt > now);
   546→  if (active.length !== sessions.length) await writeSessions(active);
   547→  return true;
   548→}

   550→export async function destroySession(token: string): Promise<void> {
   551→  const sessions = await readSessions();
   552→  const remaining = sessions.filter((s) => s.token !== token);
   553→  await writeSessions(remaining);
   554→}
